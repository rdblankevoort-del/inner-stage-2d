      <script>
          let state = {
              playerPos: 10, // % of screen
              npcPos: 80,
              tension: 20,
              playerRole: 'Modern Rebel',
              playerState: 'Anxious',
              isReversed: false
          };

          function move(dir) {
              if (dir === 'closer') state.playerPos += 10;
              else state.playerPos -= 10;

              state.playerPos = Math.min(Math.max(state.playerPos, 5), 70);

              document.getElementById('player').style.left = state.playerPos + '%';
              updateGame();
          }

          function updateGame() {
              const distance = state.npcPos - state.playerPos;
              let text = "";
              let tensionChange = 0;

              if (distance > 50) {
                  text = "The distance is too great. Silence reigns.";
                  document.getElementById('dist-text').innerText = "Distance: Far";
              } else if (distance < 15) {
                  text = "Tension is peaking! You are too close!";
                  tensionChange = 5;
                  document.getElementById('dist-text').innerText = "Distance: Intrusive";
              } else {
                  document.getElementById('dist-text').innerText = "Distance: Social";
                  if (state.playerState === 'Angry') text = "Beret: 'Your anger is a sign of weakness!'";
                  else if (state.playerState === 'Assertive') text = "Beret: 'I appreciate the directness. Proceed.'";
                  else text = "Beret: 'I am waiting for you to speak your mind.'";
              }

              document.getElementById('dialogue').innerText = text;
              state.tension = Math.min(Math.max(state.tension + tensionChange, 0), 100);
              document.getElementById('tension-bar').style.width = state.tension + '%';
          }

          window.setTone = (tone) => {
              state.playerState = tone;
              document.getElementById('state-text').innerText = tone;
              updateGame();
          };

          window.roleReversal = () => {
              state.isReversed = !state.isReversed;
              state.playerRole = state.isReversed ? 'Traditionalist' : 'Modern Rebel';
              document.getElementById('role-text').innerText = state.playerRole;

              // Visual swap
              const p = document.getElementById('player');
              const b = document.getElementById('boss');
              const tempLeft = p.style.left;
              p.style.left = b.style.left;
              b.style.left = tempLeft;

              state.playerPos = parseInt(p.style.left);
              updateGame();
          };

          // Initial call
          updateGame();
      </script>
